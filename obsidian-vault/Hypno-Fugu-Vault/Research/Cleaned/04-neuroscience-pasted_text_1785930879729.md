2. تعریف علمی استاندارد (اینترنت / peer-reviewed)
Established: هیپنوتیزم = حالت تغییر‌یافتهٔ آگاهی با (۱) توجه متمرکز، (۲) کاهش آگاهی محیطی نسبی، (۳) افزایش پاسخ به تلقین — نه خواب، نه از‌دست‌دادن کامل کنترل.

Established: تلقین‌پذیری هیپنوتیک صفت‌مانند و ابعادی است (~۱۵–۲۰٪ lows، ~۱۵–۲۰٪ highs).

Established: هیپنوتیزم به‌تنهایی معمولاً از همان درمان غیروهیپنوتیک قوی‌تر نیست؛ به‌عنوان adjunct به CBT اثر افزایشی دارد (چاقی، بی‌خوابی، اضطراب، فوبیا، فشار خون، درد، خلق).

4. نوروساینس: آنچه fMRI/EEG می‌گوید
یافته‌های Established / Strong Evidence
کاهش فعالیت قدامی Default Mode Network پس از induction در افراد high-suggestible (نه لزوماً lows).

DLPFC ↔ dACC connectivity بالاتر در highs (executive × salience).

القای selective کاهش PFC می‌تواند suggestibility را موقتاً بالا ببرد (شواهد اولیه).

import numpy as np
import pandas as pd
from sklearn.metrics import confusion_matrix, roc_curve

def youden_threshold(y_true, scores, pos_label=1):
    y_true = np.asarray(y_true)
    scores = np.asarray(scores)

    fpr, tpr, thresholds = roc_curve(y_true, scores, pos_label=pos_label)

    j = tpr - fpr
    ix = np.argmax(j)

    best_threshold = thresholds[ix]
    best_j = j[ix]
    sensitivity = tpr[ix]
    specificity = 1 - fpr[ix]

    y_pred = (scores >= best_threshold).astype(int)
    tn, fp, fn, tp = confusion_matrix(y_true, y_pred).ravel()

    return {
        "threshold": best_threshold,
        "youden_j": best_j,
        "sensitivity": sensitivity,
        "specificity": specificity,
        "tp": tp,
        "fp": fp,
        "tn": tn,
        "fn": fn
    }

def joint_youden(theta_grid, phi_grid, prior_score, pe_score, y_true):
    prior_score = np.asarray(prior_score)
    pe_score = np.asarray(pe_score)
    y_true = np.asarray(y_true)

    best = None
    rows = []

    for theta in theta_grid:
        for phi in phi_grid:
            pred = ((prior_score >= theta) & (pe_score <= phi)).astype(int)

            tp = np.sum((pred == 1) & (y_true == 1))
            fp = np.sum((pred == 1) & (y_true == 0))
            tn = np.sum((pred == 0) & (y_true == 0))
            fn = np.sum((pred == 0) & (y_true == 1))

            sens = tp / (tp + fn) if (tp + fn) else 0.0
            spec = tn / (tn + fp) if (tn + fp) else 0.0
            j = sens + spec - 1

            row = {
                "theta": theta,
                "phi": phi,
                "youden_j": j,
                "sensitivity": sens,
                "specificity": spec
            }
            rows.append(row)

            if best is None or j > best["youden_j"]:
                best = row

    return best, pd.DataFrame(rows)

پیاده‌سازی isotonic regression در پایتون

پیاده‌سازی استاندارد isotonic regression در پایتون معمولاً برای کالیبره‌کردن score یا probability خام به یک خروجی یکنواخت و بهترکالیبره انجام می‌شود، و در scikit-learn هم به‌صورت مستقیم با IsotonicRegression و هم از طریق CalibratedClassifierCV(method="isotonic") پشتیبانی می‌شود.

ساده‌ترین پیاده‌سازی
اگر یک score خام داری و یک برچسب دودویی، می‌توانی یک calibrator روی یک calibration set جدا fit کنی و بعد روی test اعمال کنی. نکتهٔ مهم این است که isotonic را نباید روی همان داده‌ای fit کنی که می‌خواهی روی آن عملکرد را قضاوت کنی، چون خطر overfitting دارد.

python
import numpy as np
from sklearn.isotonic import IsotonicRegression
from sklearn.model_selection import train_test_split
from sklearn.metrics import brier_score_loss, roc_auc_score

# y_true: labels in {0,1}
# raw_scores: uncalibrated probabilities or confidence scores

raw_scores = np.array([0.05, 0.10, 0.18, 0.22, 0.30, 0.35, 0.42, 0.51, 0.63, 0.72, 0.85, 0.93])
y_true     = np.array([0,    0,    0,    0,    0,    1,    0,    1,    1,    1,    1,    1])

scores_cal, scores_test, y_cal, y_test = train_test_split(
    raw_scores, y_true, test_size=0.4, random_state=42, stratify=y_true
)

iso = IsotonicRegression(out_of_bounds="clip")
iso.fit(scores_cal, y_cal)

calibrated_test = iso.predict(scores_test)

print("Raw Brier:", brier_score_loss(y_test, scores_test))
print("Calibrated Brier:", brier_score_loss(y_test, calibrated_test))
print("Raw AUC:", roc_auc_score(y_test, scores_test))
print("Calibrated AUC:", roc_auc_score(y_test, calibrated_test))
print("Test raw:", scores_test)
print("Test calibrated:", calibrated_test)

import numpy as np
from sklearn.isotonic import IsotonicRegression

def fit_isotonic_calibrator(raw_score, y):
    raw_score = np.asarray(raw_score, dtype=float)
    y = np.asarray(y, dtype=int)

    iso = IsotonicRegression(out_of_bounds="clip")
    iso.fit(raw_score, y)
    return iso

def apply_isotonic(iso, new_scores):
    new_scores = np.asarray(new_scores, dtype=float)
    return iso.predict(new_scores)

# example
prior_raw = np.array([0.11, 0.22, 0.25, 0.31, 0.44, 0.49, 0.57, 0.68, 0.79, 0.91])
y_event   = np.array([0,    0,    0,    0,    1,    0,    1,    1,    1,    1])

iso_prior = fit_isotonic_calibrator(prior_raw, y_event)
prior_cal = apply_isotonic(iso_prior, prior_raw)

print(prior_cal)

import numpy as np
import matplotlib.pyplot as plt
from sklearn.calibration import calibration_curve
from sklearn.metrics import brier_score_loss

# مثال
y_true = np.array([0, 0, 0, 1, 1, 1, 0, 1, 0, 1, 1, 0, 1, 0, 1])

plt.plot([0, 1], [0, 1], "--", color="gray", label="Perfect calibration")

plt.ylabel("Fraction of positives")
plt.title("Reliability Diagram")
plt.legend()
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()

python
from sklearn.datasets import make_classification
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.calibration import CalibratedClassifierCV, CalibrationDisplay
from sklearn.metrics import brier_score_loss
import matplotlib.pyplot as plt

X, y = make_classification(
    n_samples=3000,
    n_features=25,
    n_informative=10,
    random_state=42
)

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, stratify=y, random_state=42
)

base = RandomForestClassifier(n_estimators=300, random_state=42)
base.fit(X_train, y_train)
prob_raw = base.predict_proba(X_test)[:, 1]

iso = CalibratedClassifierCV(
    estimator=RandomForestClassifier(n_estimators=300, random_state=42),
    method="isotonic",
    cv=5
)
iso.fit(X_train, y_train)
prob_iso = iso.predict_proba(X_test)[:, 1]

brier_raw = brier_score_loss(y_test, prob_raw)
brier_iso = brier_score_loss(y_test, prob_iso)

fig, ax = plt.subplots(figsize=(7, 7))
CalibrationDisplay.from_predictions(
    y_test, prob_raw, n_bins=10, ax=ax,
    name=f"Raw (Brier={brier_raw:.3f})"
)
CalibrationDisplay.from_predictions(
    y_test, prob_iso, n_bins=10, ax=ax,
    name=f"Isotonic (Brier={brier_iso:.3f})"
)

ax.plot([0, 1], [0, 1], "--", color="gray", linewidth=1)
ax.set_title("Reliability Diagram: Raw vs Isotonic")
ax.grid(alpha=0.3)
plt.tight_layout()
plt.show()
