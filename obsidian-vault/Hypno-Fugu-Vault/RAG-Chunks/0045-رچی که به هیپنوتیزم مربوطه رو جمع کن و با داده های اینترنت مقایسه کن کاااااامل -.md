from sklearn.metrics import roc_curve

def youden_best_cutoff(y_true, x):
    y_true = np.asarray(y_true)
    x = np.asarray(x)

fpr, tpr, thresholds = roc_curve(y_true, x)
    j = tpr - fpr
    ix = np.argmax(j)

return {
        "cutoff": thresholds[ix],
        "youden_j": j[ix],
        "sensitivity": tpr[ix],
        "specificity": 1 - fpr[ix]
    }
