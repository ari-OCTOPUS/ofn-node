iso = CalibratedClassifierCV(
    estimator=RandomForestClassifier(n_estimators=300, random_state=42),
    method="isotonic",
    cv=5
)
iso.fit(X_train, y_train)
prob_iso = iso.predict_proba(X_test)[:, 1]

brier_raw = brier_score_loss(y_test, prob_raw)
brier_iso = brier_score_loss(y_test, prob_iso)

fig, ax = plt.subplots(figsize=(7, 7))
CalibrationDisplay.from_predictions(
    y_test, prob_raw, n_bins=10, ax=ax,
    name=f"Raw (Brier={brier_raw:.3f})"
)
CalibrationDisplay.from_predictions(
    y_test, prob_iso, n_bins=10, ax=ax,
    name=f"Isotonic (Brier={brier_iso:.3f})"
)

ax.plot([0, 1], [0, 1], "--", color="gray", linewidth=1)
ax.set_title("Reliability Diagram: Raw vs Isotonic")
ax.grid(alpha=0.3)
plt.tight_layout()
plt.show()
