#!/bin/bash
LOG=/var/log/hacash-health.log
MAX_BYTES=5242880

log() { echo [$(date '+%Y-%m-%d %H:%M:%S')] $* >> $LOG; }
[ -f $LOG ] && [ $(stat -c%s $LOG 2>/dev/null || echo 0) -gt $MAX_BYTES ] && \
    mv $LOG ${LOG}.old && touch $LOG

# 1. Service watchdog
if ! systemctl is-active --quiet hacash-fullnode; then
    log CRITICAL: fullnode down - restarting
    systemctl restart hacash-fullnode
    sleep 5
    systemctl is-active --quiet hacash-fullnode && \
        log OK: fullnode recovered || log ERROR: fullnode failed to recover
fi

# 2. RPC health
RPC_OK=0
HEIGHT=$(curl -sf --max-time 5 http://127.0.0.1:8081/query/latest/block 2>/dev/null)
if [ -n $HEIGHT ]; then
    RPC_OK=1
    BLOCK=$(echo $HEIGHT | grep -o 'height:[0-9]*' | grep -o '[0-9]*')
else
    log WARNING: RPC not responding on :8081
fi

# 3. RAM check - restart miner on Node3 if RAM critical
FREE_MB=$(awk '/MemAvailable/{print int($2/1024)}' /proc/meminfo)
if [ $FREE_MB -lt 300 ]; then
    log CRITICAL: RAM low (${FREE_MB}MB) - dropping caches
    sync && echo 3 > /proc/sys/vm/drop_caches
fi

# 4. Temperature
TEMP_RAW=$(cat /sys/class/thermal/thermal_zone0/temp 2>/dev/null || echo 0)
TEMP_C=$((TEMP_RAW / 1000))
[ $TEMP_C -gt 85 ] && log WARNING: Temp ${TEMP_C}C - throttling risk

STATUS=node=$(systemctl is-active hacash-fullnode) rpc=${RPC_OK} block=${BLOCK:-?} ram_free=${FREE_MB}MB temp=${TEMP_C}C
log STATUS: $STATUS
=== cron ===
