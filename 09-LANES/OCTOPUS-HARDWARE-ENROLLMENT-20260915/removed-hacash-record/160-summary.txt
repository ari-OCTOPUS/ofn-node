=== helper scripts ===
#!/bin/bash
MASTER="192.168.0.100:8081"
MAX_WAIT=300
ELAPSED=0
while [ $ELAPSED -lt $MAX_WAIT ]; do
    if curl -sf --max-time 3 "http://${MASTER}/" > /dev/null 2>&1; then
        echo "[$(date '+%H:%M:%S')] Master RPC ready - starting miner"
        exit 0
    fi
    echo "[$(date '+%H:%M:%S')] Waiting for master (${ELAPSED}s)..."
    sleep 10
    ELAPSED=$((ELAPSED + 10))
done
echo "[$(date '+%H:%M:%S')] Timeout - starting anyway"
exit 0
